(

       @State nvarchar(50),

       @Country nvarchar(50),

       @Action varchar(10)

As

Begin

       if @Action=’Insert’

       Begin

       Begin

            }

            catch (Exception ex)

            {

                throw ex;

            }

            return
Json(result, JsonRequestBehavior.AllowGet);

        }

    }

}
